
// Authentic-inspired box-body vehicle with articulated hood and trunk lid.
// Exterior shell is the only stock body, while audio systems remain hidden until installed.

makeBox(hoodHinge,width*.88,.048,hoodLength,0,.0,-hoodLength/2,paint);

 makeBox(hoodHinge,.023,.033,hoodLength-.11,sx*width*.36,-.02,-hoodLength/2,chrome);
}
shell.add(hoodHinge);
const trunkHinge=new T.Group();trunkHinge.name='Trunk_Hinge';

const trunkLength=half-spec.cabEnd-.07;
makeBox(trunkHinge,width*.89,.048,trunkLength,0,0,trunkLength/2,paint);
makeBox(trunkHinge,width*.80,.023,trunkLength-.09,0,-.032,trunkLength/2,black);

 makeBox(trunkHinge,.02,.025,trunkLength-.09,sx*width*.37,-.01,trunkLength/2,chrome);
}
shell.add(trunkHinge);
// Engine bay under the movable hood
const engineBay=new T.Group();engineBay.name='V8_Engine_Bay';shell.add(engineBay);
const v8=mat(0x45484b,.48,.42), engineChrome=mat(0x9a9c9d,.27,.67), hose=mat(0x151719,.84),redWire=mat(0xb22a29,.54),gold=mat(0xb4a26b,.44,.5);
const engineZ=(spec.cabStart-half)/2+.08;
makeBox(engineBay,width*.78,.12,Math.max(.65,half+spec.cabStart-.2),0,bodyTop-.25,engineZ,hose);
makeBox(engineBay,.74,.39,.96,0,bodyTop-.34,engineZ,v8);
for(const sx of [-1,1]){
 makeBox(engineBay,.245,.14,.82,sx*.37,bodyTop-.11,engineZ,engineChrome);
 for(let j=0;j<4;j++){
  makeBox(engineBay,.025,.028,.025,sx*.38,bodyTop-.03,engineZ-.30+j*.2,redWire);
 }
}

cleaner.position.set(0,bodyTop-.075,engineZ);engineBay.add(cleaner);
const airRing=new T.Mesh(new T.TorusGeometry(.29,.022,8,32),hose);
airRing.rotation.x=Math.PI/2;airRing.position.set(0,bodyTop-.005,engineZ);engineBay.add(airRing);

engineBay.add(engineFan);
// Physical stock trunk cavity. NO audio equipment is visible until a paid installation.

// Three paid options are modelled as separate, hidden installations.
const audioKits={};const audioCones=[];
const blackCab=mat(0x202124,.95),subMetal=mat(0x343940,.35,.55),subRubber=mat(0x07090b,.97),coneMat=mat(0x20262a,.58,.3),accent=mat(0xb72b34,.5,.4);
function createAudioKit(key,positions,subRadius,extra=false){
 const rack=new T.Group();rack.name='Audio_Installation_'+key;
 rack.position.set(0,bodyBottom+.19,half-.51);
 const h=key==='quad'?.56:.47,w=key==='premium'?width*.68:width*.62;
 makeBox(rack,w,h,.45,0,h*.5,0,blackCab);
 makeBox(rack,w-.04,h-.045,.022,0,h*.5,.236,subMetal);
 for(const [xx,yy,r] of positions){
  const hub=new T.Group();hub.position.set(xx,yy,.265);rack.add(hub);
  const ring=new T.Mesh(new T.TorusGeometry(r,.020,8,36),subMetal);hub.add(ring);
  const rubberRing=new T.Mesh(new T.TorusGeometry(r*.84,.033,10,36),subRubber);rubberRing.position.z=.007;hub.add(rubberRing);
  const moving=new T.Group();moving.name='Bass_Diaphragm_'+key;hub.add(moving);
  const face=new T.Mesh(new T.CylinderGeometry(r*.78,r*.41,.061,40),coneMat);face.rotation.x=Math.PI/2;face.position.z=-.018;moving.add(face);
  const cap=new T.Mesh(new T.SphereGeometry(r*.35,20,12),coneMat);cap.scale.set(1,1,.24);cap.position.z=.026;moving.add(cap);
  const trim=new T.Mesh(new T.TorusGeometry(r*.55,.005,6,32),accent);trim.position.z=.035;moving.add(trim);
  audioCones.push({mesh:moving,kit:key,baseZ:moving.position.z,mid:r<.10});
 }
 if(extra){
  for(const sx of [-1,1]){
   const amp=makeBox(rack,.10,.23,.29,sx*(w/2+.075),h*.48,0,subMetal);
   amp.name='Audio_Amplifier_'+key;
  }
 }
 rack.visible=false;root.add(rack);audioKits[key]=rack;
}
createAudioKit('dual',[[-.30,.237,.205],[.30,.237,.205]],.205);
createAudioKit('quad',[[-.33,.15,.13],[.33,.15,.13],[-.33,.41,.13],[.33,.41,.13]],.13);
createAudioKit('premium',[[-.29,.225,.19],[.29,.225,.19],[-.57,.345,.077],[.57,.345,.077]],.19,true);
